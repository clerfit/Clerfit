<script>
function searchCalculators() {
  const input = document.getElementById('searchInput').value.toLowerCase();
  const resultsContainer = document.getElementById('searchResults');
  resultsContainer.innerHTML = '';

  if (!input) return;

  const allLinks = document.querySelectorAll('.cf-links a');
  const filtered = Array.from(allLinks).filter(link =>
    link.textContent.toLowerCase().includes(input)
  );

  if (filtered.length === 0) {
    resultsContainer.innerHTML = '<p style="color:#555;">No calculators found.</p>';
    return;
  }

  filtered.forEach(link => {
    const item = document.createElement('div');
    item.innerHTML = `<a href="${link.href}">${link.textContent}</a>`;
    resultsContainer.appendChild(item);
  });
}
</script>

<!-- ================= Relative Strength JS ================= -->
<script>
function convertToKg(value, unit){
  return unit === "lbs" ? value * 0.453592 : value;
}

function calculateRelativeStrength(){
  const oneRM = parseFloat(document.getElementById("oneRM").value);
  const bodyWeight = parseFloat(document.getElementById("bodyWeight").value);
  const unit = document.getElementById("unitRelative").value;

  if(!oneRM || !bodyWeight){
    alert("Please enter both 1RM and body weight.");
    return;
  }

  const oneRMkg = convertToKg(oneRM, unit);
  const bodyWeightKg = convertToKg(bodyWeight, unit);

  const relativeStrength = oneRMkg / bodyWeightKg;

  let level;

  if(relativeStrength < 0.75){
    level = "Beginner Level";
  } else if(relativeStrength < 1.25){
    level = "Intermediate Level";
  } else if(relativeStrength < 1.75){
    level = "Advanced Level";
  } else {
    level = "Elite Level";
  }

  document.getElementById("relativeResult").innerHTML = `
    <b>Relative Strength:</b> ${relativeStrength.toFixed(2)}<br>
    <b>Strength Level:</b> ${level}
  `;
}
</script>

  <h2>Meaning & Applications</h2>

  <h3>Meaning</h3>
  <p>
    Relative Strength measures how strong you are compared to your body weight.
    It reflects strength efficiency rather than total lifting capacity.
  </p>

  <h3>Applications</h3>
  <ul>
    <li>Compare strength between individuals of different body sizes.</li>
    <li>Track progress during fat loss or muscle gain phases.</li>
    <li>Important for bodyweight sports like gymnastics, calisthenics, and martial arts.</li>
    <li>Used in athletic performance evaluation.</li>
  </ul>

  <h3>Important Note</h3>
  <p>
    Strength standards vary by exercise (bench press, squat, deadlift), age, and gender.
    This calculator gives a general reference only.
  </p>
